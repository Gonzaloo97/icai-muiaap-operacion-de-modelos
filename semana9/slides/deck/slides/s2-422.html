@app.exception_handler(RequestValidationError)
async def handle_validation(request, exc):
    fields = [
        {"field": ".".join(str(p) for p in e["loc"] if p != "body"),
         "message": e["msg"]}
        for e in exc.errors()
    ]
    return JSONResponse(
        status_code=422,
        content=error_body("validation_error", "Invalid input", fields),
    )
